/**
 * خلفية متجرك — تُنشأ لك تلقائياً (لا تعدّل شيئاً هنا). تعمل من حسابك أنت باستقلال تام.
 * تستقبل طلبات زبائن متجرك، تتحقق منهم برمز OTP على بريدهم، وتسجّلها في شيتك لتراجعها من صفحة بناء التطبيق.
 */
// بيانات متجرك تُكتب هنا تلقائياً عند الإنشاء/التحديث (لا تعدّلها يدوياً)
const APP = __APP_JSON__;
const CFG = {
  SHEET_ID: APP.sheetId,
  OWNER: APP.owner,
  OTP_REQUIRED: true,
  OTP_TTL_SEC: 1800,
  OTP_MAX_TRIES: 5,
  MIN_MAIL_QUOTA: 8,
  MAX_QTY: 10,
  PER_HOUR: 60
};
const REQ_HEADERS = ['OrderID','Timestamp','Item','Qty','Unit','Total','Customer','Phone','Email','Location','ReqTime','Lat','Lon','TxnRef','Verified','Status','DecidedAt','SheetRow','Notified','LocalTotal','Currency'];
const ORD_HEADERS = ['Timestamp','Product','Qty','Price','Total','Customer','Type','AP_Code','Location'];
const cache = CacheService.getScriptCache();

function doGet(e) { return handle_(e && e.parameter ? e.parameter : {}, true); }
function doPost(e) {
  var d = {}; try { d = JSON.parse((e && e.postData && e.postData.contents) || '{}'); } catch (err) { d = {}; }
  return handle_(d, false);
}
function out_(obj, d, fromGet) {
  var payload = JSON.stringify(obj), cb = String(d.callback || d.cb || '');
  if (fromGet && /^[A-Za-z_$][A-Za-z0-9_$]{0,40}$/.test(cb))
    return ContentService.createTextOutput(cb + '(' + payload + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  return ContentService.createTextOutput(payload).setMimeType(ContentService.MimeType.JSON);
}
function handle_(d, fromGet) {
  var res;
  try {
    var a = String(d.action || '');
    if (a === 'ping') res = { status: 'ok', app: APP.id };
    else if (a === 'config') res = config_();
    else if (a === 'unlock') res = unlock_(d);
    else if (a === 'send') res = send_(d);
    else if (a === 'verify') res = verify_(d);
    else if (a === 'decided') res = decided_(d);
    else res = { error: 'إجراء غير معروف' };
  } catch (err) { Logger.log('error: ' + err); res = { error: 'حدث خطأ — حاول مرة أخرى' }; }
  return out_(res, d, fromGet);
}

function clean_(s, max) { return String(s == null ? '' : s).replace(/[\u0000-\u001f\u007f<>]/g, ' ').replace(/\s+/g, ' ').trim().slice(0, max); }
function round_(n) { return Math.round(n * 1000) / 1000; }
function cell_(s) { s = String(s == null ? '' : s); return /^[=+\-@]/.test(s) ? "'" + s : s; }
function mailLeft_() { try { return MailApp.getRemainingDailyQuota(); } catch (e) { return 0; } }
function sheet_(name, headers) {
  var ss = SpreadsheetApp.openById(CFG.SHEET_ID), sh = ss.getSheetByName(name);
  if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); }
  else if (sh.getLastRow() === 0) sh.appendRow(headers);
  else if (name === 'Requests' && sh.getLastColumn() < headers.length) sh.getRange(1, 1, 1, headers.length).setValues([headers]);
  return sh;
}

// حالة المتجر: تجربة يوم واحد من لحظة الإنشاء، ثم يتوقف ما لم يُفتح برمز الفتح
function status_() {
  if (PropertiesService.getScriptProperties().getProperty('PAID') === '1') return 'paid';
  return Date.now() < Number(APP.trialEnds) ? 'trial' : 'locked';
}
function sha_(s) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s, Utilities.Charset.UTF_8)
    .map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
}
// بيانات الدفع: تُرسل للزبون في بريد رمز التحقق فقط (لا تُعرض في الصفحة العامة)
function payText_() {
  var p = APP.pay || {}, L = [];
  if (p.bank) L.push('البنك: ' + p.bank);
  if (p.account) L.push('رقم الحساب: ' + p.account);
  if (p.iban) L.push('IBAN: ' + p.iban);
  if (p.holder) L.push('اسم الحساب: ' + p.holder);
  if (p.note) L.push(p.note);
  return L.length ? L.join('\n') : String(APP.payInfo || '');
}
function cur_() { var p = APP.pay || {}, r = Number(p.rate); return (p.currency && r > 0) ? { code: String(p.currency), rate: r } : null; }
function local_(usd) { var c = cur_(); return c ? Math.round(usd * c.rate * 100) / 100 : ''; }
function config_() {
  var st = status_();
  if (st === 'locked') return { status: 'locked', message: 'هذا المتجر غير متاح حالياً' };
  return { status: st, name: APP.name, kind: APP.kind, bizImage: APP.bizImage, currency: cur_() ? cur_().code : '', rate: cur_() ? cur_().rate : 0,
    trialNote: st === 'trial' ? 'نسخة فعالة للبداية وقابلة للتوزيع، لكنها ليست النسخة القابلة للنشر' : '',
    items: APP.items.map(function (it, i) { return { idx: i, name: it.name, desc: it.desc, packQty: it.packQty, packLabel: it.packLabel,
      value: round_(it.price * it.packQty), localValue: local_(round_(it.price * it.packQty)), imageUrl: it.imageUrl }; }) };
}
function unlock_(d) {
  var k = 'UNL_TRIES', n = parseInt(cache.get(k) || '0', 10);
  if (n >= 5) return { error: 'محاولات كثيرة — حاول بعد ساعة' };
  try { cache.put(k, String(n + 1), 3600); } catch (e) {}
  if (sha_(String(d.code || '').trim() + ':' + APP.id) !== APP.unlockHash) return { error: 'رمز الفتح غير صحيح' };
  PropertiesService.getScriptProperties().setProperty('PAID', '1');
  return { status: 'paid' };
}
function validate_(d) {
  var st = status_(); if (st === 'locked') return { error: 'هذا المتجر غير متاح حالياً' };
  var it = APP.items[parseInt(d.idx, 10)]; if (!it) return { error: 'الصنف غير موجود' };
  var value = round_(it.price * it.packQty);
  var qty = parseInt(d.qty, 10); if (!(qty >= 1)) qty = 1;
  if (qty > CFG.MAX_QTY) return { error: 'الحد الأقصى ' + CFG.MAX_QTY + ' في الطلب الواحد' };
  var name = clean_(d.name, 60); if (name.length < 2) return { error: 'اكتب اسمك' };
  var phone = String(d.phone || '').replace(/[\s\-().]/g, '');
  if (!/^\+?[0-9]{7,15}$/.test(phone)) return { error: 'رقم الهاتف غير صحيح (أرقام فقط)' };
  var email = clean_(d.email, 80).toLowerCase();
  if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email)) return { error: 'اكتب بريداً إلكترونياً صحيحاً (يصلك عليه رمز التحقق وبيانات الدفع)' };
  var location = clean_(d.location, 120); if (!location) return { error: 'حدد الموقع' };
  var lat = Number(d.lat), lon = Number(d.lon), txn = clean_(d.txnRef, 80);
  if (txn && !/^[A-Za-z0-9\u0600-\u06FF _\-\/.#]{1,80}$/.test(txn)) return { error: 'رقم المعاملة يحتوي رموزاً غير مسموحة' };
  return { o: { item: it.name, qty: qty, unit: value, total: round_(value * qty), name: name, phone: phone, email: email,
    location: location, time: clean_(d.time, 30), txn: txn, payInfo: payText_(), biz: APP.name || '', localUnit: local_(value), localTotal: local_(round_(value * qty)), currency: cur_() ? cur_().code : '',
    lat: isFinite(lat) && Math.abs(lat) <= 90 ? lat : '', lon: isFinite(lon) && Math.abs(lon) <= 180 ? lon : '' } };
}

function send_(d) {
  var session = String(d.session || '');
  if (!/^[A-Za-z0-9_-]{16,64}$/.test(session)) return { error: 'جلسة غير صالحة' };
  var v = validate_(d); if (v.error) return v; var o = v.o;
  var rk = 'PH_' + o.phone; if (cache.get(rk)) return { error: 'انتظر دقيقة قبل إرسال طلب آخر' };
  var hn = parseInt(cache.get('HOUR') || '0', 10); if (hn >= CFG.PER_HOUR) return { error: 'المتجر مشغول حالياً، حاول لاحقاً' };
  try { cache.put(rk, '1', 60); cache.put('HOUR', String(hn + 1), 3600); } catch (e) {}
  if (!(CFG.OTP_REQUIRED && o.email && mailLeft_() > CFG.MIN_MAIL_QUOTA)) {
    var r = record_(o, false); return r.error ? r : { status: 'success', orderId: r.orderId, total: o.total, localTotal: o.localTotal, currency: o.currency, item: o.item, verified: false, payInfo: o.payInfo };
  }
  o.otp = String(Math.floor(100000 + Math.random() * 900000)); o.tries = 0;
  try { cache.put('S_' + session, JSON.stringify(o), CFG.OTP_TTL_SEC); } catch (e) { return { error: 'تعذّر بدء الطلب' }; }
  try {
    MailApp.sendEmail({ to: o.email, subject: 'رمز التحقق — ' + o.biz,
      body: 'رمز التحقق: ' + o.otp + '\n\nالمتجر: ' + o.biz + '\nالطلب: ' + o.item + ' × ' + o.qty + '\nالإجمالي بالدولار: $' + o.total +
        (o.currency ? '\nالإجمالي بـ ' + o.currency + ': ' + o.localTotal : '') +
        (o.payInfo ? '\n\nبيانات التحويل البنكي:\n' + o.payInfo : '') + '\n\nالاسم: ' + o.name + '\nالهاتف: ' + o.phone + '\nالموقع: ' + o.location + (o.time ? '\nالوقت: ' + o.time : '') });
  } catch (err) { cache.remove('S_' + session); return { error: 'تعذّر إرسال رمز التحقق إلى بريدك' }; }
  return { status: 'sent', item: o.item, qty: o.qty, total: o.total, localTotal: o.localTotal, currency: o.currency };
}
function verify_(d) {
  var session = String(d.session || ''); if (!/^[A-Za-z0-9_-]{16,64}$/.test(session)) return { error: 'جلسة غير صالحة' };
  var raw = cache.get('S_' + session); if (!raw) return { error: 'انتهت صلاحية الرمز — أعد إرسال الطلب' };
  var o = JSON.parse(raw);
  if (String(d.otp || '').trim() !== o.otp) {
    o.tries = (o.tries || 0) + 1;
    if (o.tries >= CFG.OTP_MAX_TRIES) { cache.remove('S_' + session); return { error: 'تجاوزت عدد المحاولات — أعد إرسال الطلب' }; }
    try { cache.put('S_' + session, JSON.stringify(o), CFG.OTP_TTL_SEC); } catch (e) {}
    return { error: 'رمز خاطئ' };
  }
  cache.remove('S_' + session);
  var r = record_(o, true);
  return r.error ? r : { status: 'success', orderId: r.orderId, total: o.total, localTotal: o.localTotal, currency: o.currency, item: o.item, verified: true };
}
// إشعار صاحب المتجر على جواله (Firebase عبر لاڤندرز) — لا يعطّل الطلب إن فشل
function notify_(o, id) {
  if (!APP.notifyUrl || !APP.notifyKey) return;
  try {
    UrlFetchApp.fetch(APP.notifyUrl, { method: 'post', contentType: 'text/plain;charset=utf-8', muteHttpExceptions: true,
      payload: JSON.stringify({ action: 'appb_notify', appId: APP.id, key: APP.notifyKey, customer: o.name, item: o.item, qty: o.qty, total: o.total, orderId: id }) });
  } catch (e) { Logger.log('notify: ' + e); }
}
function record_(o, verified) {
  var r = record2_(o, verified);
  if (r && r.orderId) notify_(o, r.orderId);
  return r;
}
function record2_(o, verified) {
  var lock = LockService.getScriptLock(), got = false;
  try { got = lock.tryLock(10000); } catch (e) {}
  if (!got) return { error: 'الخادم مشغول — حاول بعد لحظة' };
  try {
    var sh = sheet_('Requests', REQ_HEADERS);
    var id = 'ORD-' + Utilities.getUuid().replace(/-/g, '').slice(0, 8).toUpperCase();
    sh.appendRow([id, new Date(), cell_(o.item), o.qty, o.unit, o.total, cell_(o.name), '', cell_(o.email), cell_(o.location), cell_(o.time), o.lat, o.lon, cell_(o.txn), verified ? 'yes' : 'no', 'pending', '', '', '', o.localTotal, o.currency]);
    sh.getRange(sh.getLastRow(), 8).setNumberFormat('@').setValue(o.phone);
    if (CFG.OWNER && mailLeft_() > CFG.MIN_MAIL_QUOTA) {
      try { MailApp.sendEmail({ to: CFG.OWNER, subject: 'طلب جديد 🛍️ — ' + o.biz, body: o.name + ' — ' + o.item + ' × ' + o.qty + ' ($' + o.total + (o.currency ? ' ≈ ' + o.localTotal + ' ' + o.currency : '') + ')\nافتح صفحة بناء التطبيق لمراجعته.' }); } catch (e) {}
    }
    return { orderId: id };
  } finally { lock.releaseLock(); }
}

// إرسال بريد قبول/رفض للزبون بعد أن يسجّل المالك قراره في الشيت (مرة واحدة لكل طلب)
function decided_(d) {
  var id = String(d.orderId || ''); if (!/^ORD-[0-9A-F]{8}$/.test(id)) return { error: 'رقم الطلب غير صالح' };
  var lock = LockService.getScriptLock(), got = false;
  try { got = lock.tryLock(10000); } catch (e) {}
  if (!got) return { error: 'الخادم مشغول — حاول بعد لحظة' };
  try {
    var sh = sheet_('Requests', REQ_HEADERS), n = sh.getLastRow(); if (n < 2) return { error: 'الطلب غير موجود' };
    var v = sh.getRange(2, 1, n - 1, REQ_HEADERS.length).getValues(), r = -1, i;
    for (i = 0; i < v.length; i++) if (String(v[i][0]) === id) { r = i; break; }
    if (r < 0) return { error: 'الطلب غير موجود' };
    var x = v[r], st = String(x[15]);
    if (st !== 'accepted' && st !== 'rejected') return { error: 'لم يُتخذ قرار بعد' };
    if (String(x[18]) === 'yes') return { status: 'already' };
    var email = String(x[8] || ''); if (!email) return { status: 'no_email' };
    if (mailLeft_() <= 2) return { error: 'حصة البريد اليومية انتهت' };
    var ok = st === 'accepted', biz = APP.name || '';
    var tot = '$' + x[5] + (x[20] ? ' ≈ ' + x[19] + ' ' + x[20] : '');
    MailApp.sendEmail({ to: email, subject: (ok ? 'تم قبول طلبك ✅ — ' : 'بخصوص طلبك — ') + biz,
      body: (ok ? 'تم قبول طلبك وسيتم التواصل معك قريباً.' : 'نعتذر، لم نتمكن من قبول طلبك هذه المرة. إن كنت قد حوّلت مبلغاً فتواصل مع المتجر لاسترجاعه.') +
        '\n\nالمتجر: ' + biz + '\nرقم الطلب: ' + id + '\nالطلب: ' + x[2] + ' × ' + x[3] + '\nالإجمالي: ' + tot + (CFG.OWNER ? '\n\nللتواصل مع المتجر: ' + CFG.OWNER : '') });
    sh.getRange(r + 2, 19).setValue('yes');
    return { status: 'sent' };
  } finally { lock.releaseLock(); }
}
