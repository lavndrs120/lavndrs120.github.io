function doGet(e) {
  var p = (e && e.parameter) || {}, res;
  try { res = route_(p); } catch (err) { res = { error: String(err && err.message || err) }; }
  var out = JSON.stringify(res);
  if (p.callback && /^[A-Za-z0-9_.$]{1,60}$/.test(p.callback)) {
    return ContentService.createTextOutput(p.callback + '(' + out + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}

// حالة المؤشر: بلا trialEnds = مؤشر عضو (دائم). وإلا تجربة 24 ساعة ثم يتوقف حتى يُفتح برمز الفتح
function status_() {
  if (!M.trialEnds) return 'paid';
  if (PropertiesService.getScriptProperties().getProperty('PAID') === '1') return 'paid';
  return Date.now() < Number(M.trialEnds) ? 'trial' : 'locked';
}
function sha_(s) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s, Utilities.Charset.UTF_8)
    .map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
}
function unlock_(code) {
  var cache = CacheService.getScriptCache(), n = parseInt(cache.get('UNL') || '0', 10);
  if (n >= 5) return { error: 'محاولات كثيرة — حاول بعد ساعة' };
  try { cache.put('UNL', String(n + 1), 3600); } catch (e) {}
  if (!M.unlockHash || sha_(String(code || '').trim() + ':' + M.id) !== M.unlockHash) return { error: 'رمز الفتح غير صحيح' };
  PropertiesService.getScriptProperties().setProperty('PAID', '1');
  return { status: 'paid' };
}

function route_(p) {
  var a = p.action || 'index';
  if (a === 'ping') return { status: 'ok', market: M.id };
  if (a === 'unlock') return unlock_(p.code);
  var st = status_();
  if (st === 'locked' && a !== 'config') return { status: 'locked', error: 'انتهت التجربة المؤقتة (24 ساعة) لهذا المؤشر — يُفتح بالاشتراك في الفرانشايز أو التايكون' };
  if (a === 'plans') return plansPublic_();
  if (a === 'mkt_login') return login_(p.idToken);
  if (a === 'code_send') return codeSend_(p.email);
  if (a === 'code_verify') return codeVerify_(p.email, p.code);
  if (a === 'my_sub') return mySubInfo_(p.s);
  if (a === 'sub_start') return subStart_(p.s, p.tier);
  if (a === 'sub_confirm') return subConfirm_(p.s, p.tier, p.otp, p.ref);
  if (a === 'data' || a === 'report' || a === 'directory') return gated_(a, p.s);
  if (a === 'setup') return setup_();
  if (a === 'health') return state_(true);
  if (a === 'recalc') {
    var cc = CacheService.getScriptCache();
    if (cc.get('rc')) return { status: 'wait', message: 'انتظر دقيقة قبل إعادة الحساب' };
    cc.put('rc', '1', 60);
    return state_(false, true);
  }
  if (a === 'config') return st === 'locked' ? { status: 'locked', owner: owner_() } : { status: st, trialEnds: st === 'trial' ? M.trialEnds : 0, name: M.name, bizImage: img_(), accent: acc_(), owner: owner_() };
  return state_(false);
}

  function chg(ms) { var base = null; for (var i = pts.length - 1; i >= 0; i--) { if (now - pts[i][0] >= ms) { base = pts[i][1]; break; } } if (base == null && pts.length) base = pts[0][1]; return base > 0 ? Math.round((cur - base) / base * 10000) / 100 : 0; }
  var m30 = pts.filter(function (x) { return now - x[0] <= 30 * 86400000; }).map(function (x) { return x[1]; });
  return { nav: Number(cur.toFixed(2)), change24h: chg(86400000), change7d: chg(7 * 86400000), change30d: chg(30 * 86400000), high30d: m30.length ? Math.max.apply(null, m30) : cur, low30d: m30.length ? Math.min.apply(null, m30) : cur, businesses: rank.length, orders: total, top: rank.slice(0, 5), asOf: new Date().toISOString() };
}

// بصمة بريد المالك (الخلفية تعمل بحسابه) — تستعملها لاڤندرز لإثبات الملكية دون كشف البريد
function owner_() {

    return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, e + ':lvn-mkt', Utilities.Charset.UTF_8).map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
  } catch (err) { return ''; }

function approved_(r, ids) {
  var st = String(r[10] || '').trim().toLowerCase();
  if (st && st !== 'approved') return false;
  if (r[9] && ids && !ids[String(r[9])]) return false;
  return true;
}
function readBiz_(id) {
  var ss = SpreadsheetApp.openById(id), s = ss.getSheetByName('Orders');
  if (!s) return { net: 0 };
  var n = s.getLastRow(); if (n < 2) return { net: 0 };
  var v = s.getRange(2, 1, n - 1, 11).getValues(), cnt = {}, ids = null;
  // لا يُحتسب إلا الصف المعتمد: Status (K) = approved إن وُجد، والصف المربوط بطلب (J) يلزم أن يكون طلبه accepted في Requests
  if (v.some(function (r) { return r[9]; })) {
    var rq = ss.getSheetByName('Requests');
    if (rq) {
      ids = {};
      if (rq.getLastRow() > 1) rq.getRange(2, 1, rq.getLastRow() - 1, 16).getValues().forEach(function (r) { ids[String(r[0])] = String(r[15]) === 'accepted' ? 1 : 0; });
    }
  }
  v.forEach(function (r) {
    var item = r[1]; if (!item) return;
    if (!approved_(r, ids)) return;
    var type = r[6] || 'BUY';
    if (type === 'REDEEM') cnt[item] = Math.max(0, (cnt[item] || 0) - 1);
    else cnt[item] = (cnt[item] || 0) + 1;
  });
  var net = 0; Object.keys(cnt).forEach(function (k) { net += cnt[k]; });
  return { net: net };
}

// يحدّث أقدم الشيتات أولاً ضمن ميزانية زمنية، فيتقارب المؤشر مهما كان عدد البزنسات
function refresh_(budget) {
  var t0 = Date.now(), ss = ss_(), ts = ss.getSheetByName('Totals'), ids = list_();
  var n = ts.getLastRow(), map = {};
  if (n > 1) ts.getRange(2, 1, n - 1, 5).getValues().forEach(function (r) { map[String(r[0])] = { sum: Number(r[1]) || 0, count: Number(r[2]) || 0, at: Number(r[3]) || 0, ok: String(r[4]) === 'yes' }; });
  var order = ids.slice().sort(function (a, b) { return ((map[a] || {}).at || 0) - ((map[b] || {}).at || 0); });
  for (var i = 0; i < order.length; i++) {
    if (Date.now() - t0 > budget) break;
    var id = order[i], r = null;
    try { r = readBiz_(id); } catch (e) { r = null; }
    var prev = map[id] || { sum: 0, count: 0 };
    map[id] = r ? { sum: r.net, count: r.net, at: Date.now(), ok: true } : { sum: prev.sum, count: prev.count, at: Date.now(), ok: false };
  }
  var rows = ids.map(function (id) { var m = map[id] || { sum: 0, count: 0, at: 0, ok: false }; return [id, m.sum, m.count, m.at, m.ok ? 'yes' : 'no']; });
  if (n > 1) ts.getRange(2, 1, n - 1, 5).clearContent();
  if (rows.length) ts.getRange(2, 1, rows.length, 5).setValues(rows);
  return rows;
}

function nav_(rows) {

  return { net: net, nav: net * (M.orderValue || 4.275) };
}

function snap_(nav) {
  var ss = ss_(), sn = ss.getSheetByName('Snapshots'), now = new Date(), tz = ss.getSpreadsheetTimeZone() || Session.getScriptTimeZone() || 'Africa/Khartoum';
  var n = sn.getLastRow();
  if (n < 2) { sn.appendRow([now, 0]); n = sn.getLastRow(); }
  var last = sn.getRange(n, 1, 1, 2).getValues()[0];
  if (now.getTime() - new Date(last[0]).getTime() >= 55 * 60000) { sn.appendRow([now, nav]); n = sn.getLastRow(); }
  if (n > 5000) { sn.deleteRows(2, 1000); n = sn.getLastRow(); }
  var cnt = Math.min(30, Math.max(1, n - 1));
  var rows = sn.getRange(Math.max(2, n - 29), 1, cnt, 2).getValues();
  var values = rows.map(function (r) { return Number(r[1]) || 0; }), rets = [], i;
  for (i = 1; i < values.length; i++) if (values[i - 1] > 0) rets.push((values[i] - values[i - 1]) / values[i - 1]);
  var avg = rets.length ? rets.reduce(function (a, b) { return a + b; }, 0) / rets.length : 0;
  var variance = rets.length ? rets.reduce(function (a, b) { return a + Math.pow(b - avg, 2); }, 0) / rets.length : 0;
  var vol = Math.sqrt(variance * 252) * 100;
  var day = Utilities.formatDate(now, tz, 'yyyy-MM-dd');
  var today = rows.filter(function (r) { return Utilities.formatDate(new Date(r[0]), tz, 'yyyy-MM-dd') === day; }).map(function (r) { return Number(r[1]) || 0; });
  var hi = today.length ? Math.max.apply(null, today) : nav, lo = today.length ? Math.min.apply(null, today) : nav;
  var prev = rows.length > 1 ? Number(rows[rows.length - 2][1]) || 0 : nav;
  var ch = nav - prev, pct = prev > 0 ? ch / prev * 100 : 0;
  return { NAV: nav.toFixed(2), change: ch.toFixed(2), changePercent: pct.toFixed(2), dayHigh: hi.toFixed(2), dayLow: lo.toFixed(2), volatility: vol.toFixed(2) };
}

function state_(healthOnly, force) {
  var lock = LockService.getScriptLock(), got = false;
  try { got = lock.tryLock(force ? 20000 : 8000); } catch (e) {}
  var cache = CacheService.getScriptCache(), key = healthOnly ? 'h' : 'i', hit = cache.get(key);
  if (hit && !healthOnly && !force) return JSON.parse(hit);
  var rows;
  try { rows = refresh_(got ? (force ? 120000 : BUDGET_MS) : 0); } finally { if (got) lock.releaseLock(); }
  var ok = rows.filter(function (r) { return r[4] === 'yes'; }).length, orders = 0;
  rows.forEach(function (r) { orders += Number(r[2]) || 0; });
  var calc = nav_(rows), nav = calc.nav, res;
  if (healthOnly) return { status: 'ok', market: M.id, businesses: rows.length, readable: ok, orders: orders };
  res = { market: M.name, mid: M.id, plans: plans_().on ? 1 : 0, image: img_(), accent: acc_(), index: snap_(nav), businesses: rows.length, orders: orders, timestamp: new Date().toISOString() };
  cache.put('i', JSON.stringify(res), 240);
  if (force) { res.status = 'ok'; res.readable = ok; }
  return res;
}
