function doGet(e) {
  var p = (e && e.parameter) || {}, res;
  try { res = route_(p); } catch (err) { res = { error: String(err && err.message || err) }; }
  var out = JSON.stringify(res);
  if (p.callback && /^[A-Za-z0-9_.$]{1,60}$/.test(p.callback)) {
    return ContentService.createTextOutput(p.callback + '(' + out + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}

// حالة المؤشر: بلا trialEnds = مؤشر عضو (دائم). وإلا تجربة 24 ساعة ثم يتوقف حتى يُفتح برمز الفتح
function status_() {
  if (!M.trialEnds) return 'paid';
  if (PropertiesService.getScriptProperties().getProperty('PAID') === '1') return 'paid';
  return Date.now() < Number(M.trialEnds) ? 'trial' : 'locked';
}
function sha_(s) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s, Utilities.Charset.UTF_8)
    .map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
}
function unlock_(code) {
  var cache = CacheService.getScriptCache(), n = parseInt(cache.get('UNL') || '0', 10);
  if (n >= 5) return { error: 'محاولات كثيرة — حاول بعد ساعة' };
  try { cache.put('UNL', String(n + 1), 3600); } catch (e) {}
  if (!M.unlockHash || sha_(String(code || '').trim() + ':' + M.id) !== M.unlockHash) return { error: 'رمز الفتح غير صحيح' };
  PropertiesService.getScriptProperties().setProperty('PAID', '1');
  return { status: 'paid' };
}

function route_(p) {
  var a = p.action || 'index';
  if (a === 'ping') {
    // يُشغّل الصلاحيتين فعلياً: إن لم تُمنح بعد يظهر طلب الموافقة هنا
    var mail = 'no', ext = 'no';
    try { MailApp.getRemainingDailyQuota(); mail = 'ok'; } catch (e) {}
    try { UrlFetchApp.fetch('https://oauth2.googleapis.com/tokeninfo?id_token=x', { muteHttpExceptions: true }); ext = 'ok'; } catch (e) {}
    return { status: 'ok', market: M.id, mail: mail, external: ext };
  }
  if (a === 'unlock') return unlock_(p.code);
  var st = status_();
  if (st === 'locked' && a !== 'config') return { status: 'locked', error: 'انتهت التجربة المؤقتة (24 ساعة) لهذا المؤشر — يُفتح بالاشتراك في الفرانشايز أو التايكون' };
  if (a === 'plans') return plansPublic_();
  if (a === 'mkt_login') return login_(p.idToken);
  if (a === 'mkt_handoff_new') return handoffNew_(p.s);
  if (a === 'mkt_handoff_use') return handoffUse_(p.c);
  if (a === 'code_send') return codeSend_(p.email);
  if (a === 'code_verify') return codeVerify_(p.email, p.code);
  if (a === 'my_sub') return mySubInfo_(p.s);
  if (a === 'sub_start') return subStart_(p.s, p.tier);
  if (a === 'sub_confirm') return subConfirm_(p.s, p.tier, p.otp, p.ref);
  if (a === 'data' || a === 'report' || a === 'directory') return gated_(a, p.s);
  if (a === 'dash_public') return dashPublic_();
  if (a === 'dash_data') return dashData_(p.s, p.vt);
  if (a === 'dash_oracle') return dashOracle_(p.s, p.period);
  if (a === 'dash_month') return dashMonth_(p.s, p.month);
  if (a === 'setup') return setup_();
  if (a === 'health') return state_(true);
  if (a === 'recalc') {
    var cc = CacheService.getScriptCache();
    if (cc.get('rc')) return { status: 'wait', message: 'انتظر دقيقة قبل إعادة الحساب' };
    cc.put('rc', '1', 60);
    return state_(false, true);
  }
  if (a === 'config') return st === 'locked' ? { status: 'locked', owner: owner_() } : { status: st, trialEnds: st === 'trial' ? M.trialEnds : 0, name: M.name, mid: M.id, bizImage: img_(), accent: acc_(), owner: owner_() };
  return state_(false);
}

}
